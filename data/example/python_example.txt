A Route Request TLV prompts the receiver to send an update for a given prefix, or a full route table dump.

Fields:

Type
Set to 9 to indicate a Route Request TLV.
Length
The length of the body in octets, exclusive of the Type and Length fields.
AE
The encoding of the Prefix field. The value 0 specifies that this is a request for a full route table dump (a wildcard request).
Plen
The length in bits of the requested prefix.
Prefix
The prefix being requested. This field's size is Plen/8 rounded upwards.
A Request TLV prompts the receiver to send an update message (possibly a retraction) for the prefix specified by the AE, Plen, and Prefix fields, or a full dump of its route table if AE is 0 (in which case Plen must be 0 and Prefix is of length 0). A Request TLV with AE set to 0 and Plen not set to 0 MUST be ignored.

This TLV is self-terminating and allows sub-TLVs.

def parse_packet_route_request_tlv(
  current_result: dict, remaining_input: bytes
) -> tuple[dict, bytes]:
  """
  Parse a Route Request TLV (Type 9) from RFC specification.

  Args:
      current_result: Dictionary containing already parsed fields
      remaining_input: Bytes to parse

  Returns:
      Tuple of (parsed_fields_dict, remaining_bytes)

  Raises:
      ValueError: If the input data doesn't conform to RFC specifications
  """
  # Minimum length check: Type (1) + Length (1) + AE (1) + Plen (1) = 4 bytes
  if len(remaining_input) < 4:
    raise ValueError("Route Request TLV requires at least 4 bytes")

  # Parse fixed header fields
  tlv_type = remaining_input[0]
  length = remaining_input[1]
  ae = remaining_input[2]
  plen = remaining_input[3]

  # Validate Type field
  if tlv_type != 9:
    raise ValueError(f"Expected Type 9 for Route Request TLV, got {tlv_type}")

  # Validate Length field consistency
  expected_body_length = 2 + (plen + 7) // 8  # AE (1) + Plen (1) + Prefix (variable)
  if length != expected_body_length:
    raise ValueError(
      f"Length field {length} doesn't match expected body length {expected_body_length}"
    )

  # Validate AE and Plen constraints
  if ae == 0:
    # Wildcard request: Plen must be 0 and Prefix must be empty
    if plen != 0:
      raise ValueError("AE=0 (wildcard request) requires Plen=0")
    if length != 2:  # AE (1) + Plen (1)
      raise ValueError("AE=0 (wildcard request) requires Length=2")
    prefix = b""
    remaining_bytes = remaining_input[4:]
  else:
    # Specific prefix request
    prefix_length_bytes = (plen + 7) // 8  # Calculate bytes needed for prefix

    # Check if we have enough bytes for the prefix
    if len(remaining_input) < 4 + prefix_length_bytes:
      raise ValueError(
        f"Insufficient bytes for prefix: need {prefix_length_bytes} bytes"
      )

    # Extract prefix
    prefix = remaining_input[4 : 4 + prefix_length_bytes]
    remaining_bytes = remaining_input[4 + prefix_length_bytes :]

  sub_tlvs = []

  # handle subtlvs
  while len(remaining_bytes) > 0:
    sub_tlv_result, remaining_bytes = parse_packet_subtlv({}, remaining_bytes)
    sub_tlvs.append(sub_tlv_result)

  # Build result dictionary
  result = {
    "type": tlv_type,
    "length": length,
    "ae": ae,
    "plen": plen,
    "prefix": prefix,
    "sub_tlvs": sub_tlvs,
  }

  # Update current_result with parsed fields
  current_result.update(result)

  return current_result, remaining_bytes

   The BFD Control packet has a Mandatory Section and an optional

   +-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+-+

def parse_packet_bfd_control_packet(
  current_result: dict, remaining_input: bytes
) -> tuple[dict, bytes]:
  """
  Parse a BFD Control Packet from RFC specification.

  Args:
      current_result: Dictionary containing already parsed fields
      remaining_input: Bytes to parse

  Returns:
      Tuple of (parsed_fields_dict, remaining_bytes)

  Raises:
      ValueError: If the input data doesn't conform to RFC specifications
  """
  # Minimum length check for mandatory section: 24 bytes
  if len(remaining_input) < 24:
    raise ValueError(
      "BFD Control Packet requires at least 24 bytes for mandatory section"
    )

  # Parse the first byte (Vers and Diag)
  first_byte = remaining_input[0]
  vers = (first_byte & 0xE0) >> 5  # Bits 7-5 (3 bits)
  diag = first_byte & 0x1F  # Bits 4-0 (5 bits)

  # Parse the second byte (flags and Detect Mult)
  second_byte = remaining_input[1]
  sta = (second_byte & 0xC0) >> 6  # Bits 7-6 (2 bits)
  p_flag = (second_byte & 0x20) >> 5  # Bit 5 (1 bit)
  f_flag = (second_byte & 0x10) >> 4  # Bit 4 (1 bit)
  c_flag = (second_byte & 0x08) >> 3  # Bit 3 (1 bit)
  a_flag = (second_byte & 0x04) >> 2  # Bit 2 (1 bit)
  d_flag = (second_byte & 0x02) >> 1  # Bit 1 (1 bit)
  m_flag = second_byte & 0x01  # Bit 0 (1 bit)
  detect_mult = remaining_input[2]
  length = remaining_input[3]

  # Validate Version field
  if vers != 1:
    raise ValueError(f"Expected Version 1 for BFD Control Packet, got {vers}")

  # Validate Diagnostic field
  if diag > 8 and diag < 31:
    raise ValueError(f"Diagnostic value {diag} is reserved for future use")

  # Validate State field
  if sta > 3:
    raise ValueError(f"Invalid State value {sta}, must be 0-3")

  # Validate Multipoint bit
  if m_flag != 0:
    raise ValueError("Multipoint bit (M) must be zero")

  # Validate Length field
  if length < 24:
    raise ValueError(f"Length field {length} is less than minimum 24 bytes")
  if length != len(remaining_input):
    raise ValueError(
      f"Length field {length} does not match actual input length {len(remaining_input)}"
    )

  # Parse discriminator fields (4 bytes each)
  my_discriminator = int.from_bytes(remaining_input[4:8], byteorder="big")
  your_discriminator = int.from_bytes(remaining_input[8:12], byteorder="big")

  # Validate My Discriminator
  if my_discriminator == 0:
    raise ValueError("My Discriminator must be nonzero")

  # Validate Your Discriminator and Packet State
  if sta in (2, 3) and your_discriminator == 0:
    raise ValueError(
      "Your Discriminator must be nonzero when State is Init or Up"
    )

  # Parse interval fields (4 bytes each)
  desired_min_tx_interval = int.from_bytes(remaining_input[12:16], byteorder="big")
  required_min_rx_interval = int.from_bytes(remaining_input[16:20], byteorder="big")
  required_min_echo_rx_interval = int.from_bytes(
    remaining_input[20:24], byteorder="big"
  )

  # Validate Desired Min TX Interval
  if desired_min_tx_interval == 0:
    raise ValueError("Desired Min TX Interval must not be zero")

  # Calculate remaining bytes after mandatory section
  remaining_bytes = remaining_input[24:length]

  # Handle authentication section if present
  auth_data = None
  auth_type = None
  auth_len = None

  if a_flag:
    # Check if we have enough bytes for authentication header
    if len(remaining_bytes) < 2:
      raise ValueError(
        "Authentication Present bit set but insufficient bytes for auth header"
      )

    auth_type = remaining_bytes[0]
    auth_len = remaining_bytes[1]

    # Validate Auth Type
    if auth_type == 0 or (auth_type > 5 and auth_type < 256):
      raise ValueError(f"Auth Type {auth_type} is reserved")

    # Validate Auth Len
    if auth_len < 2:
      raise ValueError(f"Auth Len {auth_len} must be at least 2")
    if auth_len > len(remaining_bytes):
      raise ValueError(f"Auth Len {auth_len} exceeds available auth data")

    # Extract authentication data
    auth_section = None
    match auth_type:
      case 1:  # Simple Password
        auth_section, remaining_bytes = parse_packet_simple_password_auth_section(
          {}, remaining_bytes[:auth_len]
        )
      case 2:  # Keyed MD5
        auth_section, remaining_bytes = parse_packet_keyed_md5_auth_section(
          {}, remaining_bytes[:auth_len]
        )
      case 3:  # Meticulous Keyed MD5
        auth_section, remaining_bytes = parse_packet_meticulous_keyed_md5_auth_section(
          {}, remaining_bytes[:auth_len]
        )
      case 4:  # Keyed SHA1
        auth_section, remaining_bytes = parse_packet_keyed_sha1_auth_section(
          {}, remaining_bytes[:auth_len]
        )
      case 5:  # Meticulous Keyed SHA1
        auth_section, remaining_bytes = parse_packet_meticulous_keyed_sha1_auth_section(
          {}, remaining_bytes[:auth_len]
        )
      case _:
        raise ValueError(f"Unsupported Auth Type {auth_type}")
    # remaining_bytes = remaining_bytes[auth_len:]

  # Build result dictionary
  result = {
    "version": vers,
    "diagnostic": diag,
    "state": sta,
    "poll": bool(p_flag),
    "final": bool(f_flag),
    "control_plane_independent": bool(c_flag),
    "authentication_present": bool(a_flag),
    "demand": bool(d_flag),
    "multipoint": bool(m_flag),
    "detect_mult": detect_mult,
    "length": length,
    "my_discriminator": my_discriminator,
    "your_discriminator": your_discriminator,
    "desired_min_tx_interval": desired_min_tx_interval,
    "required_min_rx_interval": required_min_rx_interval,
    "required_min_echo_rx_interval": required_min_echo_rx_interval,
  }

  # Add authentication fields if present
  if a_flag:
    result.update(
      {"auth_type": auth_type, "auth_len": auth_len, "auth_section": auth_section}
    )

  # Update current_result with parsed fields
  current_result.update(result)

  return current_result, remaining_bytes
